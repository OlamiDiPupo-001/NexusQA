1""" 

2Database setup using SQLAlchemy 2.0's typed ORM style. Starts on SQLite 

3for local development (zero setup — it's just a file), and swaps to

8import os

11from sqlalchemy import create_engine 

12from sqlalchemy.orm import DeclarativeBase, Mapped, mapped_column, sessionmaker

14DATABASE_URL = os.getenv("NEXUSQA_DB_URL", "sqlite:///./nexusqa.db")

16connect_args = {"check_same_thread": False} if "sqlite" in DATABASE_URL else {}

25class OrderRecord(Base):

39class ProcessedWebhookEvent(Base):

50class StockRecord(Base): 

51 """A single limited-stock item, used ONLY to give Layer 3's

56 __tablename__ = "stock"

58 id: Mapped[int] = mapped_column(primary_key=True)

87def seed_stock(quantity: int = 3) -> None: 

88 """(Re)sets the flash-sale stock item to a known quantity. Tests call 

89 this directly before each run so leftover state never affects the 

90 next test's result.""" 

91 db = SessionLocal() 

92 existing = db.query(StockRecord).filter_by(item_name="flash_sale_item").first() 

93 if existing: 

94 existing.quantity = quantity 

95 else: 

96 db.add(StockRecord(item_name="flash_sale_item", quantity=quantity)) 

97 db.commit()
